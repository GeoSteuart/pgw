const STR = {strong:{c:"#166534",bg:"#DCFCE7",l:"Strong Pathway"},applicable:{c:"#1D4ED8",bg:"#DBEAFE",l:"Applicable"},possible:{c:"#92400E",bg:"#FEF3C7",l:"Evidence Needed"}};

let state = { step:0, sel:[], ev:{}, status:"none", t75:false, wriisc:false, reg:false, catF:"All" };

function cats(){ let s=new Set(C.map(c=>c.cat)); return ["All",...[...s].sort()]; }

function getPaths(id){
  let e=state.ev[id]||{}, info=C.find(c=>c.id===id), p=[], n=[];
  let dx=e.diagnosed, nx=e.nexus, lb=e.labWork, isM=MUCMI.includes(id);
  if(dx&&nx) p.push({name:"Direct Service Connection",str:"strong",d:"Diagnosis and nexus opinion on file. Typically the strongest pathway. Evaluated on its own merits without relying on the Gulf War presumptive framework."});
  else if(dx&&!nx) p.push({name:"Direct Service Connection",str:"possible",d:"Diagnosis exists but no nexus opinion linking the condition to service. A medical professional would need to provide a written opinion that the condition is at least as likely as not related to military service."});
  if(!dx){
    p.push({name:"Undiagnosed Illness (\u00a7 3.317(a)(2)(i)(A))",str:"applicable",d:"No formal diagnosis attributed. May qualify as an undiagnosed illness if the condition cannot be attributed to any known clinical diagnosis by history, physical examination, and laboratory tests. No nexus opinion required under this pathway."});
    if(!lb) n.push("The regulation requires that the condition cannot be attributed to a known clinical diagnosis by history, physical examination, AND laboratory tests. If lab testing has not been performed, the examiner cannot properly make this determination. Request appropriate testing before the C&P examination.");
  }
  if(dx&&isM) p.push({name:"MUCMI (\u00a7 3.317(a)(2)(i)(B))",str:"strong",d:info.label+" is specifically named in the regulation as a medically unexplained chronic multisymptom illness. This pathway exists for diagnosed conditions lacking conclusive pathophysiology or etiology. No nexus opinion required."});
  else if(dx&&!isM) p.push({name:"MUCMI (\u00a7 3.317(a)(2)(i)(B))",str:"possible",d:"Diagnosed condition not specifically named in the regulation, but the MUCMI pathway accommodates any diagnosed condition lacking conclusive pathophysiology or etiology, characterized by overlapping symptoms and signs, with features including fatigue, pain, disability out of proportion to physical findings, and inconsistent laboratory abnormalities."});
  if(id==="ptsd") n.push("PTSD has its own framework under 38 C.F.R. Section 3.304(f) and is not typically filed under Gulf War presumptive. Combat veterans who served in a war zone have relaxed stressor verification.");
  if(id==="als") n.push("ALS is a presumptive condition for all veterans with 90 or more days of continuous active service under 38 U.S.C. Section 1118. Rated at 100% from date of diagnosis.");
  if(id==="tinnitus"||id==="hearing") n.push("The Hearing Loss and Tinnitus DBQ is restricted and not publicly available. Only VA or contract examiners can complete it.");
  return {p,n};
}

function downloadFile(content, filename, mimeType){
  let blob = new Blob([content], {type: mimeType+";charset=utf-8"});
  let url = URL.createObjectURL(blob);
  let a = document.createElement("a");
  a.href = url; a.download = filename;
  document.body.appendChild(a); a.click();
  document.body.removeChild(a);
  URL.revokeObjectURL(url);
}
